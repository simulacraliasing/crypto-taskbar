cmake_minimum_required(VERSION 3.20)
project(CryptoTaskbar VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(crypto_core INTERFACE)
target_include_directories(crypto_core INTERFACE src vendor)
if(WIN32)
  enable_language(RC)
  add_executable(CryptoTaskbar WIN32 src/main.cpp resources/app.rc)
  target_include_directories(CryptoTaskbar PRIVATE resources)
  target_link_libraries(CryptoTaskbar PRIVATE crypto_core winhttp gdiplus dwmapi shell32 comctl32 advapi32)
  target_compile_definitions(CryptoTaskbar PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  add_executable(network_tests EXCLUDE_FROM_ALL tests/network_tests.cpp)
  target_link_libraries(network_tests PRIVATE crypto_core winhttp)
  target_compile_definitions(network_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MINGW)
    target_link_options(CryptoTaskbar PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_compile_options(CryptoTaskbar PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(network_tests PRIVATE -static -static-libgcc -static-libstdc++)
  elseif(MSVC)
    target_compile_options(CryptoTaskbar PRIVATE /utf-8 /W4 /EHsc)
    set_property(TARGET CryptoTaskbar PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE crypto_core)
if(MINGW)
  target_link_options(core_tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME core COMMAND core_tests)
